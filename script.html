economia <- left_join(desempleo, ipc, by = "mes")

#Crear tabla con las elecciones

elecciones <- data.frame(
  eleccion = c(1989, 1993, 1993, 1999, 1999, 2005, 2005, 2009, 2009,
               2013, 2013, 2013, 2017, 2017, 2017, 2021, 2021, 2021, 2025, 2025, 2025),
  vuelta   = c("primera", "primaria", "primera", "primera", "segunda", "primera", "segunda", "primera", "segunda",

               "primaria", "primera", "segunda", "primaria", "primera", "segunda"),
  fecha    = as.Date(c("1989-12-14", "1993-05-23", "1993-12-11", "1999-12-12", "2000-01-16",

                       "2013-06-30", "2013-11-17", "2013-12-15",
                       "2017-07-02", "2017-11-19", "2017-12-17",

)

#Aca calculamos el promedio de los indicadores economicos que usamos de los 12 meses anteriories

elecciones$desempleo <- NA
elecciones$ipc <- NA

for (i in 1:nrow(elecciones)) {
  mes_eleccion <- floor_date(elecciones$fecha[i], "month")
  ventana <- filter(economia, mes >= mes_eleccion - months(12), mes < mes_eleccion)
  elecciones$desempleo[i] <- round(mean(ventana$desempleo), 2)
  elecciones$ipc[i] <- round(mean(ventana$ipc), 2)
}

#Cargamos la base de esandalos y renombramos las columnas, tambien dejamos todos los nombres en minusculas para que sea mas facil

escandalos <- read_excel("data/base.xlsx", sheet = "escandalos")

names(escandalos) <- c("anio", "mes", "candidato", "genero", "escandalo",
                       "tipo", "magnitud", "cercania", "fecha",
                       "descripcion", "link")
escandalos$candidato <- trimws(escandalos$candidato)
escandalos$candidato <- tolower(escandalos$candidato)
escandalos$candidato <- chartr("áéíóúñü", "aeiounu", escandalos$candidato)

#Los meses sin escandalos les dejamos magnitud vacía y creamos la variable de cada mes para poder indentificar los escandalos dentro del rango de meses

escandalos$magnitud[is.na(escandalos$magnitud)] <- 0
escandalos$mes_fecha <- as.Date(paste(escandalos$anio, escandalos$mes, "01", sep = "-"))

#Cargamos las pestañas de candidaturas y candidaturas

candidaturas <- read_excel("data/base.xlsx", sheet = "candidaturas")
names(candidaturas) <- c("candidato", "anio", "vuelta", "partido", "sector",
                         "resultado", "mecanismo", "votos")

candidatos <- read_excel("data/base.xlsx", sheet = "candidatos")
names(candidatos) <- c("candidato", "genero", "pais")

candidaturas$candidato <- trimws(candidaturas$candidato)
candidaturas$candidato <- tolower(candidaturas$candidato)
candidaturas$candidato <- chartr("áéíóúñü", "aeiounu", candidaturas$candidato)

candidatos$candidato <- trimws(candidatos$candidato)
candidatos$candidato <- tolower(candidatos$candidato)
candidatos$candidato <- chartr("áéíóúñü", "aeiounu", candidatos$candidato)

#Creamos la variable si gano o no

candidaturas$gano <- ifelse(candidaturas$resultado == "ganador", 1, 0)

candidaturas$vuelta[candidaturas$vuelta == "1ra vuelta"] <- "primera"
candidaturas$vuelta[candidaturas$vuelta == "2da vuelta"] <- "segunda"

#Creamos la variable que indican el año de la eleccion, las segundas vueltas que se realizan en enero del año siguiente las asignamos al año anterior por temas de orden

candidaturas$eleccion <- candidaturas$anio
candidaturas$eleccion[candidaturas$anio == 2000] <- 1999
candidaturas$eleccion[candidaturas$anio == 2006] <- 2005
candidaturas$eleccion[candidaturas$anio == 2010] <- 2009

#Escándalos por candidatura, cada candidatura suma solo los escándalos ocurridos en los 12 meses anteriores a su votación.

escandalos_meses<- escandalos[, c("candidato", "mes_fecha", "escandalo", "magnitud")]

cruce <- candidaturas |>
  select(candidato, eleccion, vuelta) |>
  left_join(elecciones[, c("eleccion", "vuelta", "fecha")], by = c("eleccion", "vuelta")) |>
  left_join(escandalos_meses, by = "candidato")
## Warning in left_join(left_join(select(candidaturas, candidato, eleccion, : Detected an unexpected many-to-many relationship between `x` and `y`.

## ℹ Row 164 of `y` matches multiple rows in `x`.

cruce <- cruce |>
  filter(mes_fecha >= floor_date(fecha, "month") - months(11),
         mes_fecha <= floor_date(fecha, "month"))

resumen <- cruce |>
  group_by(candidato, eleccion, vuelta) |>
  summarise(
    n_escandalos = sum(escandalo),   # cantidad de escándalos
    magnitud_max = max(magnitud),    # gravedad del escándalo más grave
    .groups = "drop"
  )

resumen$escandalo <- if_else(resumen$n_escandalos > 0, 1, 0)

#Aqui unimos todas las tablas que hicimos a la base de candidaturas

base <- candidaturas %>%
  left_join(candidatos, by = "candidato") %>%
    left_join(resumen, by = c("candidato", "eleccion", "vuelta")) |>
  left_join(elecciones, by = c("eleccion", "vuelta"))

base$votos <- base$votos * 100

#Escala ideológica: de -3 (extrema izquierda) a 3 (extrema derecha), con el centro en 0, (centro/populista se considera centro)

base <- base |>
  mutate(ideologia = case_when(
    sector == "extrema izquierda" ~ -3,
    sector == "izquierda"         ~ -2,
    sector == "centroizquierda"   ~ -1,
    sector == "centro"            ~  0,
    sector == "centro/populista"  ~  0,
    sector == "centroderecha"     ~  1,
    sector == "derecha"           ~  2,
    sector == "extrema derecha"   ~  3))

#Candidaturas sin meses de escándalos en su ventana
base$n_escandalos[is.na(base$n_escandalos)] <- 0
base$magnitud_max[is.na(base$magnitud_max)] <- 0
base$escandalo[is.na(base$escandalo)] <- 0

#Carrera: la votación en que cada candidato compitió contra sus rivales (en 2013, 2017 y 2021 hubo dos primarias el mismo día, una por coalición)

base <- base |>
  mutate(carrera = paste(eleccion, vuelta),
         carrera = case_when(
           vuelta == "primaria" & candidato %in% c("pablo longueira", "andres allamand") ~ "2013 primaria alianza",
           vuelta == "primaria" & candidato %in% c("beatriz sanchez", "alberto mayol") ~ "2017 primaria frente amplio",
           vuelta == "primaria" & candidato %in% c("gabriel boric", "daniel jadue") ~ "2021 primaria apruebo dignidad",
           TRUE ~ carrera))

#Veremos los modelos ahora, primero vemos modelos lineales que explican el porcentaje de votos de cada candidatura, el 4 es de caracter multiplicativo

# Modelo 1: solo variables de interes
m1_votos <- lm(votos ~ n_escandalos + genero,
               data = base)

# Modelo 2: se agregan variables politicas
m2_votos <- lm(votos ~ n_escandalos + genero + magnitud_max + ideologia + vuelta,
               data = base)

# Modelo 3: se agregan controles economicos
m3_votos <- lm(votos ~ n_escandalos + genero + magnitud_max + ideologia + vuelta + desempleo + ipc,
               data = base)

# Modelo 4: modelo multiplicativo (interacción escándalos × género)
m4_votos <- lm(votos ~ n_escandalos * genero + magnitud_max + ideologia + vuelta + desempleo + ipc,
               data = base)

#Aqui usamos un modelo logit condicional (clogit) que explican la probabilidad de ganar la vuelta comparando a cada candidatura solo son sus rivales de la misma carrera, strata(carrera) significa que cada candidatura se comara solo con sus rivales de la carrera.

# Modelo 5: solo variables de interes
c1_gano <- clogit(gano ~ n_escandalos + genero + strata(carrera),
                  data = base)

# Modelo 6: se agregan magnitud e ideologia
c2_gano <- clogit(gano ~ n_escandalos + genero + magnitud_max + ideologia + strata(carrera),
                  data = base)

# Modelo 7: modelo multiplicativo (interacción escandalos × genero)
c3_gano <- clogit(gano ~ n_escandalos * genero + magnitud_max + ideologia + strata(carrera),
                  data = base)
